# Web framework. Tip ipuçlarından otomatik istek doğrulaması üretir — M3'te
# gelen ingest payload'larını elle kontrol etmek yerine Pydantic modelleriyle
# tanımlayacağız, bozuk payload endpoint koduna hiç ulaşmayacak.
fastapi==0.115.6

# ASGI sunucusu — FastAPI kendi başına HTTP konuşmaz, uvicorn konuşur.
# [standard] eklentisi: httptools (hızlı HTTP ayrıştırma) + uvloop (hızlı olay
# döngüsü). Ingest yolu çok sayıda küçük POST alacağı için ikisi de işe yarar.
uvicorn[standard]==0.34.0

# Supabase'in PostgREST arayüzüne yazmak için HTTP istemcisi. async/await
# desteklediği için FastAPI'nin olay döngüsünü bloklamaz; tek bağlantı havuzu
# uygulama ömrü boyunca açık kalır (supabase_client.py).
httpx==0.28.1

# Dashboard'un gönderdiği Supabase access token'ını doğrular (auth.py).
# Token'ın imzasını, süresini ve hedef alanlarını kontrol eder; `sub` alanı
# account_id olarak okunur.
pyjwt==2.13.0

# pyjwt'nin ES256 (eliptik eğri) imzasını doğrulayabilmesi için gereken kripto
# motoru. Saf Python'da yapılamayan tek adım budur: bu paket kurulu değilse
# pyjwt ES256 token'ı doğrulamayı reddeder.
#
# Derleyici gerektirmez: PyPI'da `cp311-abi3` etiketli hazır wheel yayınlanıyor
# ve Python 3.12'nin (Dockerfile + CI) kullandığı sürüm budur.
cryptography==50.0.0
